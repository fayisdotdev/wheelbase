-- create table vehicles

create table public.vehicles (
  id uuid default uuid_generate_v4() primary key,

  -- Relation
  user_auth_uuid uuid not null references public.users(auth_uuid) on delete cascade,

  -- Basic vehicle info
  owner_name text not null,
  vehicle_name text not null,
  vehicle_number text not null,
  vehicle_year text not null,

  -- Documents validity
  insurance_starts date not null,
  insurance_ends date not null,
  pollution_starts date not null,
  pollution_ends date not null,

  -- Maintenance
  battery text,
  alignment text,
  service_km text not null,
  notes text,

  -- Options
  need_notification boolean default false,
  shared_with boolean default false,

  -- Metadata
  vehicle_id text not null unique,  -- your unique string (custom, like VIN/UUID)
  vehicle_added_by text not null,   -- store uploader’s name/email/etc.
  image_url text,

  uploaded_at timestamptz default now()
);

-- rls policies
alter table public.vehicles enable row level security;

-- Allow insert only if user_auth_uuid matches the logged-in user
create policy "Users can insert their own vehicles"
on public.vehicles for insert
with check (auth.uid() = user_auth_uuid);

-- Allow select only their own vehicles
create policy "Users can view their own vehicles"
on public.vehicles for select
using (auth.uid() = user_auth_uuid);
