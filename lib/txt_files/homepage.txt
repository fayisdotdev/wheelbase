class AuthProvider extends ChangeNotifier {
  final SupabaseClient _supabase = Supabase.instance.client;

  Session? _session;
  User? _user;
  UserProfileModel? _profile;

  User? get user => _user;
  UserProfileModel? get profile => _profile;
  bool get isLoggedIn => _session != null;

  AuthProvider() {
    _session = _supabase.auth.currentSession;
    _user = _supabase.auth.currentUser;

    if (_user != null) {
      fetchUserProfile(_user!.id).then((p) {
        _profile = p;
        notifyListeners();
      });
    }
  }

  Future<UserProfileModel?> fetchUserProfile(String authUuid) async {
    final response = await _supabase
        .from('users')
        .select()
        .eq('auth_uuid', authUuid)
        .maybeSingle();

    if (response != null) {
      return UserProfileModel.fromMap(response);
    }
    return null;
  }

  /// Signup
  Future<String?> signUp({
    required String email,
    required String password,
    required String name,
    required String phone,
  }) async {
    try {
      final response = await _supabase.auth.signUp(
        email: email,
        password: password,
      );

      if (response.user == null) {
        return "Signup failed";
      }

      final cleanEmail = email.replaceAll(RegExp(r'[^a-zA-Z0-9@.]'), '');
      final customUid = cleanEmail;

      await _supabase.from('users').insert({
        'auth_uuid': response.user!.id,
        'email': email,
        'name': name,
        'phone': phone,
        'uuid': customUid,
        'password': password,
      });

      _session = response.session;
      _user = response.user;
      _profile = UserProfileModel(
        authUuid: response.user!.id,
        uuid: customUid,
        email: email,
        name: name,
        phone: phone,
        password: password,
        createdAt: DateTime.now(),
      );
      notifyListeners();
      return null;
    } catch (e) {
      return e.toString();
    }
  }

  /// Login
  Future<String?> login({
    required String email,
    required String password,
  }) async {
    try {
      final response = await _supabase.auth.signInWithPassword(
        email: email,
        password: password,
      );

      if (response.user == null) return "Login failed";

      _session = response.session;
      _user = response.user;
      _profile = await fetchUserProfile(response.user!.id);

      notifyListeners();
      return null;
    } catch (e) {
      return e.toString();
    }
  }

  Future<void> logout() async {
    await _supabase.auth.signOut();
    _session = null;
    _user = null;
    _profile = null;
    notifyListeners();
  }
}
